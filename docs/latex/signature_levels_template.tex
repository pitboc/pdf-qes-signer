\usepackage{needspace}
\titleformat{\section}
  {\needspace{8\baselineskip}\raggedright\large\bfseries\color{qesblue}}{\thesection}{0.6em}{}

  {\needspace{4\baselineskip}\raggedright\normalsize\bfseries\color{qesblue}}{\thesubsection}{0.5em}{}

  {\needspace{3\baselineskip}\raggedright\normalsize\bfseries\color{qesblue!80!black}}{\thesubsubsection}{0.5em}{}
\titlespacing*{\section}{0pt}{16pt}{6pt}
\titlespacing*{\subsection}{0pt}{10pt}{3pt}
\titlespacing*{\subsubsection}{0pt}{7pt}{2pt}

%% Shaded environment (pandoc code blocks with a language, overrides the
%% pandoc default if already defined by the macros above; otherwise a
%% fallback definition so documents WITHOUT code blocks don't fail with
%% "Environment Shaded undefined" -- \highlighting-macros is only emitted by
%% pandoc when the document contains at least one code block)
\makeatletter
\@ifundefined{Shaded}{\newenvironment{Shaded}{}{}}{}
\makeatother
\renewenvironment{Shaded}{%
  \begin{mdframed}[%
    backgroundcolor=codebg,
    linecolor=gray!40,
    linewidth=0.4pt,
    innerleftmargin=10pt,
    innerrightmargin=10pt,
    innertopmargin=5pt,
    innerbottommargin=5pt,
    skipabove=6pt,
    skipbelow=4pt,
  ]}{\end{mdframed}}
\usepackage{mdframed}

%% Verbatim environment (pandoc code blocks without a language)
\DefineVerbatimEnvironment{verbatim}{Verbatim}{
  fontsize=\small,
  xleftmargin=10pt,
  xrightmargin=10pt,
  frame=single,
  framerule=0.4pt,
  framesep=5pt,
  rulecolor=\color{gray!40},
  fillcolor=\color{codebg},
}

    innerrightmargin=8pt,
